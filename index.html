function submitComplaint(){ const type=document.getElementById('problemType').value; const place=document.getElementById('problemPlace').value.trim(); const description=document.getElementById('problemDescription').value.trim(); if(!type || !place || !description){ alert('Please fill all fields'); return; } const id='CC'+Date.now().toString().slice(-8); const complaints=JSON.parse(localStorage.getItem('ccComplaints')||'[]'); complaints.push({ id,type,place,description, status:'Pending', date:new Date().toLocaleString() }); localStorage.setItem('ccComplaints',JSON.stringify(complaints)); document.getElementById('complaintMessage').textContent='Complaint submitted successfully!'; document.getElementById('complaintId').textContent='Complaint ID: '+id; }
